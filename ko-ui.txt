function logout(){ localStorage.removeItem("nwtoken"); location.reload(); }

async function doLogin(){
  const id = $("gid").value.trim(), pw = $("gpw").value;
  if(!id || !pw) return;
  try {
    const r = await fetch("/api/login", { method:"POST", headers:{"content-type":"application/json"},
      body: JSON.stringify({ id, pw }) });
    const d = await r.json();
    if(!r.ok || d.error) { $("gateMsg").textContent = d.error || "로그인 실패"; return; }
    localStorage.setItem("nwtoken", d.token);
    $("gpw").value = "";
    boot();
  } catch(e){ $("gateMsg").textContent = String(e.message||e); }
}

/* ── 날짜·시간 표기 ── */
function fmt(s){
  if(!s) return "";
  const d = new Date(s);
  if(isNaN(d)) return String(s).slice(0,16);
  const p = n => String(n).padStart(2,"0");
  return d.getFullYear()+"-"+p(d.getMonth()+1)+"-"+p(d.getDate())+" "+p(d.getHours())+":"+p(d.getMinutes());
}

async function load(){
  $("body").innerHTML = '<div class="muted" style="padding:20px 0">불러오는 중…</div>';
  $("msg").innerHTML = "";

    $("body").innerHTML = "";

async function fetchJson(url){
  const r = await fetch(url, { cache:"no-store", headers: headers() });
  if(r.status === 401){ showGate("다시 로그인해 주세요."); throw new Error("로그인이 필요합니다"); }
  const d = await r.json();
  // 저장소 미연결은 "고장" 이 아니라 "아직 설정 안 됨" 이다.
  // 여기서 던져 버리면 무엇을 해야 하는지 안내할 기회를 잃는다.
  if(d.error && d.historyEnabled !== false) throw new Error(d.error);
  return d;
}

async function loadAdmin(){
  const qs = new URLSearchParams({ view: VIEW });
  if($("q").value.trim()) qs.set("q", $("q").value.trim());
  if($("days").value !== "0") qs.set("days", $("days").value);
  if($("by").value) qs.set("by", $("by").value);
  // 서버 기록은 UTC 다. 한국 아침 8시 발송은 UTC 로 전날이라, 시차를 넘겨야 날짜가 안 밀린다.
  if(VIEW === "daily") qs.set("tz", String(new Date().getTimezoneOffset()));
  const d = await fetchJson("/api/admin?" + qs);

  if(d.historyEnabled === false){
    $("body").innerHTML = setupGuide(d);
    return "";
  }
  if(VIEW === "daily")   return renderDaily(d);
  if(VIEW === "summary") return renderSummary(d);
  if(VIEW === "sent")    return renderSent(d);
  if(VIEW === "blocked") return renderBlocked(d);
  if(VIEW === "conversations") return renderConversations(d);
  if(VIEW === "pipeline") return renderPipeline(d);
  return "";
}

/* 기록이 하나도 없을 때. 동기화는 자동으로 돌지 않으므로, 빈 표만 보여주면
   "고장인가" 와 "아직 안 눌렀나" 를 구분할 수 없다. 무엇을 하면 되는지 같이 띄운다. */
/* 자동 동기화가 마지막으로 언제 돌았는지. 숫자가 낡았는지 화면에서 바로 알아야 한다 */
function cronNote(d){
  const c = d.cron;
  if(!c) return '자동 동기화(매일 07:00 KST)가 아직 돈 기록이 없습니다. ' +
           '눌러서 <b>2026-05-01 이후</b> 발송·회신을 지금 바로 가져오세요.';
  const when = fmt(c.at);
  const rem = c.reminders;
  return '마지막 자동 동기화 <b>'+esc(when)+'</b>' +
    (c.complete ? '' : ' <span style="color:#a8801f">(전원을 다 못 돌아 다음 실행이 이어받습니다)</span>') +
    (c.totals && c.totals.failed ? ' <span style="color:#a23b2e">· 실패 '+c.totals.failed+'명</span>' : '') +
    (rem && (rem.sent || rem.replied) ? ' <span class="muted">· 리마인드 '+(rem.sent||0)+'건 발송'+
      (rem.replied ? ', 회신 도착으로 '+rem.replied+'건 종료' : '')+'</span>' : '') +
    ' <span class="muted">· 매일 07:00(KST) 자동 실행</span>';
}

function emptyNote(d){
  if((d.totals && d.totals.sent) || (d.rows && d.rows.length)) return "";
  return '<div class="note"><b>아직 기록이 없습니다.</b><br />' +
    '자동 동기화는 <b>매일 07:00(KST)</b>에 돕니다. 아직 한 번도 안 돌았다면, 담당자별 탭의 ' +
    '<b>[🔄 메일함에서 동기화]</b> 로 <b>2026-05-01 이후</b> 발송·회신을 지금 바로 가져올 수 있습니다.<br />' +
    '<span class="muted">담당자 수에 따라 몇 분 걸립니다. 이 도구로 새로 보내는 메일은 기다리지 않아도 바로 잡힙니다.</span></div>';
}

function truncNote(d){

    : "";
}

/* ── 일별 발송 ──
   시간축을 따라 크기를 비교하는 일이므로 세로 막대다. 계열은 둘(발송·중복 시도)이라
   범례를 항상 두고, 값은 점마다 찍지 않고 마우스를 올렸을 때만 보여준다. */
const BAND = 26, BARW = 9, GAP = 2, H = 190, PADT = 12, PADB = 26, PADL = 34;

// 위 모서리만 둥근 막대. rect 의 rx 는 네 모서리를 다 깎아 바닥이 떠 보인다.
function barPath(x, y, w, h, r){
  if(h <= 0) return "";
  const rr = Math.min(r, w/2, h);
  return "M"+x+","+(y+h)+" L"+x+","+(y+rr)+" Q"+x+","+y+" "+(x+rr)+","+y+
         " L"+(x+w-rr)+","+y+" Q"+(x+w)+","+y+" "+(x+w)+","+(y+rr)+" L"+(x+w)+","+(y+h)+" Z";
}

// 비율 표기 — 분모가 0이면 숫자 대신 줄표를 낸다 (0% 는 거짓말이다)
function pct(a, b){ return b ? Math.round(a/b*100) + "%" : "—"; }

function niceMax(n){
  if(n <= 4) return Math.max(n, 4);
  const p = Math.pow(10, Math.floor(Math.log10(n)));
  return Math.ceil(n / (p/2)) * (p/2);
}

function renderDaily(d){
  const rows = d.rows || [];
  if(!rows.length){ $("body").innerHTML = emptyNote(d); return ""; }

  const totalSent = rows.reduce((s,r)=>s+r.sent,0);
  const totalBlocked = rows.reduce((s,r)=>s+r.blocked,0);
  const totalReplied = rows.reduce((s,r)=>s+(r.replied||0),0);
  const busiest = rows.reduce((a,b)=> b.sent > a.sent ? b : a, rows[0]);
  const activeDays = rows.filter(r=>r.sent>0).length;
  const max = niceMax(Math.max(1, ...rows.map(r=>Math.max(r.sent, r.replied||0))));

  const W = PADL + rows.length * BAND + 8;
  const plotH = H - PADT - PADB;
  const y = v => PADT + plotH - (v / max) * plotH;

  // 가로선 4개 — 표면에서 한 단계만 떨어진 회색, 눈에 띄지 않게
  let g = "";
  for(let i=0;i<=4;i++){
    const v = max * i / 4, yy = y(v);
    g += '<line class="gridline" x1="'+PADL+'" y1="'+yy+'" x2="'+W+'" y2="'+yy+'" />' +
         '<text class="tick" x="'+(PADL-7)+'" y="'+(yy+3.5)+'" text-anchor="end">'+Math.round(v)+'</text>';
  }

  // 날짜 라벨은 겹치지 않을 만큼만
  const every = Math.max(1, Math.ceil(rows.length / 12));

  let bars = "", hits = "";
  rows.forEach((r,i) => {
    const x0 = PADL + i * BAND;
    const bx = x0 + (BAND - (BARW*2 + GAP)) / 2;
    if(r.sent)    bars += '<path d="'+barPath(bx, y(r.sent), BARW, PADT+plotH-y(r.sent), 4)+'" fill="var(--sent)" />';
    if(r.replied) bars += '<path d="'+barPath(bx+BARW+GAP, y(r.replied), BARW, PADT+plotH-y(r.replied), 4)+'" fill="var(--replied)" />';

    if(i % every === 0){
      bars += '<text class="tick" x="'+(x0+BAND/2)+'" y="'+(H-8)+'" text-anchor="middle">'+
              esc(r.date.slice(5).replace("-","/"))+'</text>';
    }
    // 손이 닿는 곳은 막대보다 넓게 — 하루 칸 전체
    hits += '<rect class="band" x="'+x0+'" y="'+PADT+'" width="'+BAND+'" height="'+plotH+'" />' +
            '<rect class="hit" x="'+x0+'" y="'+PADT+'" width="'+BAND+'" height="'+plotH+
            '" data-i="'+i+'" />';
  });

  $("body").innerHTML =
    '<div class="cards">' +
      '<div class="card"><div class="l">기간 내 발송</div><div class="n mono">'+totalSent+'</div></div>' +
      '<div class="card"><div class="l">발송한 날</div><div class="n mono">'+activeDays+'<span style="font-size:14px;color:#999">일</span></div></div>' +
      '<div class="card"><div class="l">회신</div><div class="n mono">'+totalReplied+
        '</div><div class="l" style="margin-top:2px">'+pct(totalReplied, totalSent)+' 회신율</div></div>' +
      '<div class="card"><div class="l">가장 많은 날</div><div class="n mono">'+busiest.sent+
        '</div><div class="l" style="margin-top:2px">'+esc(busiest.date)+'</div></div>' +
      '<div class="card"><div class="l">중복으로 막힘</div><div class="n mono">'+totalBlocked+'</div></div>' +
    '</div>' +
    '<div class="viz">' +
      '<div class="legend"><span><i style="background:var(--sent)"></i>발송</span>' +
      '<span><i style="background:var(--replied)"></i>회신</span>' +
      '<span class="muted" style="margin-left:auto">막대에 마우스를 올리면 그날 내역이 보입니다</span></div>' +
      '<div class="vizwrap" id="vizWrap">' +
        '<svg width="'+W+'" height="'+H+'" role="img" aria-label="일별 아웃리치 발송 건수">' +
        '<g>'+g+'</g><g>'+bars+'</g><g>'+hits+'</g></svg>' +
        '<div class="tip" id="tip"></div>' +
      '</div>' +
    '</div>' +
    // 차트로 못 읽는 값은 표로 읽는다
    '<div class="scroll" style="margin-top:6px"><table><thead><tr>' +
      '<th>날짜</th><th>발송</th><th>회신</th><th>중복 시도</th><th>담당자별</th>' +
    '</tr></thead><tbody>' +
    rows.slice().reverse().filter(r=>r.sent||r.blocked||r.replied).map(r =>
      '<tr><td class="mono">'+esc(r.date)+'</td>' +
      '<td class="mono">'+r.sent+'</td>' +
      '<td class="mono">'+(r.replied ? '<b style="color:#158a61">'+r.replied+'</b>' : '<span class="muted">0</span>')+'</td>' +
      '<td class="mono">'+(r.blocked ? '<span class="tag hold">'+r.blocked+'</span>' : '<span class="muted">0</span>')+'</td>' +
      '<td>'+esc(Object.keys(r.by||{}).map(k => k+" "+r.by[k]).join(" · "))+'</td></tr>').join("") +
    '</tbody></table></div>';

  wireTips(rows);
  return totalSent + "건";
}

function wireTips(rows){
  const wrap = $("vizWrap"), tip = $("tip");
  if(!wrap || !tip) return;
  wrap.querySelectorAll(".hit").forEach(el => {
    el.addEventListener("mouseenter", () => {
      const r = rows[Number(el.dataset.i)];
      const who = Object.keys(r.by||{});
      tip.innerHTML = '<b>'+esc(r.date)+'</b><br />' +
        '<span class="k">발송</span> <b>'+r.sent+'</b>건' +
        (r.replied ? '<br /><span class="k">회신</span> <b>'+r.replied+'</b>건' : '') +
        (r.blocked ? '<br /><span class="k">중복으로 막힘</span> <b>'+r.blocked+'</b>건' : '') +
        (who.length ? '<br />'+who.map(k => '<span class="k">'+esc(k)+'</span> '+r.by[k]).join('<br />') : '');
      tip.classList.add("on");
      const x = Number(el.getAttribute("x")) + BAND/2 - wrap.scrollLeft;
      tip.style.left = Math.max(4, Math.min(x - tip.offsetWidth/2, wrap.clientWidth - tip.offsetWidth - 4)) + "px";
      tip.style.top = "4px";
    });
    el.addEventListener("mouseleave", () => tip.classList.remove("on"));
  });
}

/* ── 메일함에서 동기화 ──
   담당자별로 ① 보낸편지함 → 발송 이력, ② 받은편지함 → 회신 기록 순으로 채운다.
   전원을 한 요청에 몰면 함수 제한시간(60초)을 넘기므로 **한 사람씩 나눠서** 부르고,
   진행 상황을 그때그때 보여준다 — 오래 걸리는 작업이 말없이 도는 게 제일 나쁘다.

   보낸편지함에는 아웃리치가 아닌 메일이 섞여 있다. 사내 도메인 수신자와
   여러 명에게 보낸 메일(6명 이상)은 서버가 자동으로 걸러낸다. */
const SYNC_SINCE = "2026-05-01";
let SYNC_RESULT = null;   // 마지막 동기화 결과 — 화면을 다시 그려도 남아 있어야 한다

async function syncAll(){
  const btn = $("syncBtn"), msg = $("syncMsg");
  if(!btn) return;
  btn.disabled = true;

  const people = [...$("by").options].slice(1).map(o => ({ email:o.value, name:o.textContent }));
  if(!people.length){ msg.textContent = "등록된 담당자가 없습니다 (NW_ACCOUNTS)."; btn.disabled = false; return; }

  // 함수가 시간을 넘기면 Vercel 이 JSON 이 아니라 오류 **페이지**를 돌려준다.
  // 그걸 그대로 JSON 으로 읽으면 "Unexpected token 'A'" 같은 쓸모없는 말이 뜬다 —
  // 무엇이 잘못됐는지 사람이 알아볼 수 있게 상태 코드로 풀어서 알려준다.
  const call = async (url, body) => {
    const r = await fetch(url, {
      method:"POST", cache:"no-store",
      headers: Object.assign({ "content-type":"application/json" }, headers()),
      body: JSON.stringify(body)
    });
    if(r.status === 401){ showGate("다시 로그인해 주세요."); throw new Error("로그인이 필요합니다"); }
    const text = await r.text();
    let d = null;
    try { d = JSON.parse(text); } catch(_) {
      if(r.status === 504 || /timeout|FUNCTION_INVOCATION/i.test(text))
        throw new Error("시간 초과 — 메일이 많아 한 번에 못 끝냈습니다. 다시 누르면 이어서 처리됩니다.");
      throw new Error("서버 오류 (HTTP "+r.status+")");
    }
    if(d.error) throw new Error(d.error);
    return d;
  };

  let sent = 0, replies = 0;
  const failed = [];
  for(let i = 0; i < people.length; i++){
    const p = people[i];
    SYNC_RESULT = '<b>'+(i+1)+'/'+people.length+'</b> '+esc(p.name)+' 메일함을 읽는 중… ' +
      '<span class="muted">(인원에 따라 몇 분 걸릴 수 있습니다)</span>';
    msg.innerHTML = SYNC_RESULT;
    try {
      const b = await call("/api/backfill", { user:p.email, since:SYNC_SINCE, dryRun:false });
      sent += (b.result && b.result.imported) || 0;
      const r = await call("/api/replies", { user:p.email, since:SYNC_SINCE });
      replies += (r.totals && r.totals.found) || 0;
    } catch(e){
      if(/로그인/.test(String(e.message))) { btn.disabled = false; return; }
      failed.push(p.name + " — " + String(e.message||e));
    }
  }

  SYNC_RESULT = (failed.length ? '동기화 끝 (일부 실패)' : '동기화 완료') +
    ' · 새 발송 <b>'+sent+'</b>건 · 새 회신 <b>'+replies+'</b>건' +
    ' <span class="muted">(2026-05-01 이후, 이미 있던 건 건너뜀)</span>' +
    (failed.length ? '<br /><span style="color:#a23b2e">실패: '+esc(failed.join(" / "))+'</span>' : '');
  msg.innerHTML = SYNC_RESULT;
  btn.disabled = false;
  load();
}

  // accounts 모드에서는 발신 계정을 서버가 정한다 — 브라우저에는 앱 비밀번호가 없다
  if(AUTH_MODE === "accounts") return ME;
  try { return JSON.parse(localStorage.getItem("nwacct")||"null"); } catch(e){ return null; }
}
function openAcct(){
  if(AUTH_MODE === "accounts") return;   // 로그인 계정이 곧 발신 계정
  const a = acct() || {};
  $("acctName").value = a.name||""; $("acctTitle").value = a.title||"";
  $("acctEmail").value = a.email||""; $("acctPass").value = a.password||"";
  $("acctErr").textContent = "";
  $("acctModal").style.display="flex"; $("acctEmail").focus();
}
function closeAcct(){ $("acctModal").style.display="none"; }
function saveAcct(){
  const a = { name:$("acctName").value.trim(), title:$("acctTitle").value.trim(),
              email:$("acctEmail").value.trim().toLowerCase(), password:$("acctPass").value };
  if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(a.email)){ $("acctErr").textContent="이메일 형식을 확인하세요."; return; }
  if(CFG.allowedDomains && CFG.allowedDomains.length){
    const d = a.email.split("@")[1];
    if(CFG.allowedDomains.indexOf(d) === -1){
      $("acctErr").textContent = "허용된 발신 도메인이 아닙니다: @" + CFG.allowedDomains.join(" / @");
      return;
    }
  }
  if(!a.password){ $("acctErr").textContent="외부 앱 비밀번호를 입력하세요."; return; }
  if(!a.name) a.name = a.email.split("@")[0];
  localStorage.setItem("nwacct", JSON.stringify(a));
  closeAcct(); paintWho(); renderPreview();
}
function clearAcct(){ localStorage.removeItem("nwacct"); closeAcct(); paintWho(); renderPreview(); }
function paintWho(){
  const a = acct(); const b = $("whoBtn");
  if(a && a.email){
    b.textContent = "✉️ " + a.email;
    b.classList.remove("unset");
    b.style.cursor = AUTH_MODE === "accounts" ? "default" : "pointer";
    b.title = AUTH_MODE === "accounts" ? "로그인 계정으로 발송됩니다" : "클릭해서 발신 계정 변경";
  } else {
    b.textContent = AUTH_MODE === "accounts" ? "로그인 필요" : "발신 계정 미설정";
    b.classList.add("unset");
  }
}

function setDeal(d){
  DEAL = d;
  ["paid","affiliate","both","none"].forEach(k => $("dt-"+k).classList.toggle("on", k===d));

/* ── TikTok Shop 상품 선택기 ──
   자격증명(TTS_*)이 없으면 /api/products 가 501 을 주고, 그때는 버튼 자체를 숨긴다.
   상품 선택은 편의 기능일 뿐이라 없어도 나머지는 그대로 동작해야 한다. */
let PRODUCTS = null;
let PICKED_PID = "";      // 고른 제품의 TikTok Shop 상품 ID — 매출 상위 영상 조회에 쓴다
let PRODUCT_IMG_DATA = ""; // 업로드한 제품 이미지 (data URL) — URL 이 없을 때 파일 첨부
let VIDEOS_READY = false; // 영상 시트가 설정된 배포에서만 버튼을 띄운다
let VIDEOS_WHY = "";      // 안 되면 그 이유 — 조용히 사라지면 원인을 짚을 수 없다

async function loadProducts(force){
  const msg = $("pickMsg");
  msg.textContent = "불러오는 중…";
  $("pickGrid").innerHTML = "";
  try {
    const qs = [];
    if($("pickAll").checked) qs.push("all=1");
    if(force) qs.push("fresh=1");
    const r = await fetch("/api/products" + (qs.length ? "?"+qs.join("&") : ""),
      { cache:"no-store", headers: authHeaders() });
    const data = await r.json();
    if(r.status === 401){ showGate("다시 로그인해 주세요."); return false; }
    if(r.status === 501){ PRODUCTS = null; $("pickBar").style.display="none"; closePicker(); return false; }
    if(data.error) throw new Error(data.error + (data.hint ? "\n" + data.hint : ""));
    PRODUCTS = data.products || [];
    msg.textContent = PRODUCTS.length
      ? PRODUCTS.length + "개" + ($("pickAll").checked ? "" : " (판매중)")
      : "상품이 없습니다.";
    renderPicker();
    return true;
  } catch(e){
    msg.innerHTML = '<span style="color:#c0392b">'+esc(String(e.message||e)).replace(/\n/g,"<br />")+'</span>';
    return false;
  }
}

/* jpeg 변형이 없는 CDN 템플릿이면 원본(webp) URL 로 한 번 더 시도하고,
   그것도 실패해야 "이미지 없음" 으로 바꾼다. */
function thumbFail(img){
  const alt = img.getAttribute("data-alt");
  if(alt && img.src !== alt){ img.removeAttribute("data-alt"); img.src = alt; return; }
  img.outerHTML = '<div class="noimg">이미지 없음</div>';
}

function renderPicker(){
  if(!PRODUCTS) return;
  const q = ($("pickSearch").value || "").trim().toLowerCase();
  /* 상품명과 상품 ID 둘 다로 찾는다 — 담당자끼리 ID 로 주고받는 경우가 많다 */
  const list = q ? PRODUCTS.filter(p => p.title.toLowerCase().includes(q) || p.id.includes(q)) : PRODUCTS;
  $("pickGrid").innerHTML = list.slice(0, 120).map(p =>
    '<button type="button" class="pcard" onclick="choose('+PRODUCTS.indexOf(p)+')">' +
    (p.image
      ? '<img class="thumb" src="'+esc(p.image)+'" alt="" loading="lazy"' +
        ' data-alt="'+esc(p.imageOrig||"")+'" onerror="thumbFail(this)" />'
      : '<div class="noimg">이미지 없음</div>') +
    '<div class="nm">'+esc(p.title)+'</div>' +
    '<div class="mt">'+esc(p.currency+" "+p.price)+
      (p.stock>0 ? ' · 재고 '+p.stock : ' · <span class="oos">품절</span>')+'</div>' +
    '<div class="pid">'+esc(p.id)+
      (p.status && p.status.toUpperCase()!=="ACTIVATE" ? ' · <span class="oos">'+esc(p.status)+'</span>' : '')+
    '</div>' +
    '</button>'
  ).join("") || '<div style="color:#999;font-size:12.5px">검색 결과가 없습니다.</div>';
}

/* 검색창 입력 처리.
   ID 처럼 생긴 숫자를 쳤는데 불러온 목록에 없으면 — 판매중이 아니거나
   목록 밖의 상품일 수 있으므로 — 그 ID 하나만 서버에 직접 물어본다. */
let idTimer = 0;
function onPickSearch(){
  renderPicker();
  clearTimeout(idTimer);
  const q = ($("pickSearch").value || "").trim();
  if(!/^\d{6,}$/.test(q)) return;
  if(PRODUCTS && PRODUCTS.some(p => p.id === q)) return;
  idTimer = setTimeout(() => lookupId(q), 400);
}

async function lookupId(id){
  const msg = $("pickMsg");
  msg.textContent = "상품 ID " + id + " 조회 중…";
  try {
    const r = await fetch("/api/products?id=" + encodeURIComponent(id),
      { cache:"no-store", headers: authHeaders() });
    const d = await r.json();
    if(r.status === 401){ showGate("다시 로그인해 주세요."); return; }
    if(d.error) throw new Error(d.error);
    const p = (d.products || [])[0];
    if(!p){ msg.textContent = "그 ID 의 상품을 찾지 못했습니다."; return; }
    if(!PRODUCTS) PRODUCTS = [];
    if(!PRODUCTS.some(x => x.id === p.id)) PRODUCTS.unshift(p);
    msg.textContent = "ID 로 찾음" + (p.status ? " · " + p.status : "");
    renderPicker();
  } catch(e){
    msg.innerHTML = '<span style="color:#c0392b">'+esc(String(e.message||e))+'</span>';
  }
}

function choose(i){
  const p = PRODUCTS[i];
  if(!p) return;
  $("product").value = p.title;
  if(PRODUCT_IMG_DATA) setImgData("");   // 목록에서 고른 상품은 자체 이미지가 있으니 첨부는 비운다
  $("productImage").value = p.image || "";
  // 링크가 비어 있어도 상품 ID 만은 남긴다 — 여기서 pid 를 되찾기 때문
  $("productUrl").value = p.url || (p.id ? "https://shop.tiktok.com/view/product/" + p.id : "");
  syncPid();
  paintPicked(p);
  closePicker();
  renderPreview();
}

/* ── 고른 제품의 상품 ID ──
   pid 를 따로 저장하지 않고 **제품 링크에서 되찾는다.** 링크는 이미 초안에 저장되고
   그 안에 상품 ID 가 들어 있다. 저장을 두 군데 두면 새로고침한 뒤 둘이 어긋나서,
   제품은 그대로 채워져 있는데 영상 버튼만 사라지는 일이 생긴다(실제로 그랬다).
   링크를 손으로 붙여 넣은 경우에도 그대로 동작한다. */
function pidFromUrl(u){
  const m = String(u||"").match(/\/product\/(\d{6,})/);
  return m ? m[1] : "";
}

function syncPid(){
  const next = pidFromUrl(val("productUrl"));
  if(next !== PICKED_PID){
    PICKED_PID = next;
    TOPVIDS = null;                       // 제품이 바뀌면 앞 제품의 영상 목록은 버린다
    $("vidList").style.display = "none";
    $("vidList").innerHTML = "";
    $("vidMsg").textContent = "";
  }
  paintVidBar();
}

let TOPVIDS = null;

async function loadTopVideos(){
  const pid = PICKED_PID;
  const msg = $("vidMsg"), box = $("vidList");
  if(!pid){ msg.textContent = "먼저 위에서 제품을 고르세요."; return; }

  box.style.display = "none";

    const r = await fetch("/api/videos?pid=" + encodeURIComponent(pid) + "&limit=30",
      { cache:"no-store", headers: authHeaders() });
    if(r.status === 401){ showGate("다시 로그인해 주세요."); return; }
    const d = await r.json();

    msg.innerHTML = '<span style="color:#c0392b">'+esc(String(e.message||e))+'</span>';
  }
}

/* 이미 칸에 들어가 있는 영상은 다시 넣지 않도록 표시해 둔다 */
function videoSlots(){ return ["v1","v2","v3"].map(k => ({ u:$(k+"u"), l:$(k+"l") })); }

function renderTopVideos(){
  const inUse = videoSlots().map(s => s.u.value.trim()).filter(Boolean);
  $("vidList").innerHTML = '<div class="vids">' + TOPVIDS.map((v,i) => {
    const used = inUse.indexOf(v.url) >= 0;

      '<span class="h">@'+esc(v.handle)+'</span>' +
      '<span class="v">'+(used ? "넣음" : "고르기")+'</span>' +
      '</button>';
  }).join("") + '</div>';

function pickVideo(i){
  const v = TOPVIDS && TOPVIDS[i];
  if(!v) return;
  const slots = videoSlots();
  if(slots.some(s => s.u.value.trim() === v.url)) return;   // 이미 넣은 것
  const empty = slots.find(s => !s.u.value.trim());
  if(!empty){ $("vidMsg").textContent = "영상 칸 3개가 다 찼습니다. 하나를 비우고 다시 고르세요."; return; }
  empty.u.value = v.url;
  // 라벨에는 GMV 금액을 쓰지 않는다 — 크리에이터 핸들만 남긴다
  empty.l.value = "@" + v.handle;
  renderTopVideos();
  renderPreview();
}

/* ── 명단 파싱 (탭 = 시트 붙여넣기, 콤마 = CSV) ── */
/* 헤더 인식 — 실제 시트는 "E-Mail", "Creator Name", "TikTok Handle" 처럼 표기가 제각각이라
   ① 영숫자·한글만 남겨 정규화한 뒤 ② 아래 규칙을 위에서부터 부분일치로 검사한다.
   순서가 중요하다: "creator email" 은 email(수신주소)이지 creator(이름)가 아니다. */
const COLRULES = [
  [/mail|이메일|메일|주소/,                 "to"],
  [/handle|핸들|tiktok|틱톡|username|계정/,  "handle"],
  [/commission|커미션|수수료|율/,            "commission"],
  [/amount|금액|fee|budget|price|단가|보수|비용/, "amount"],
  [/why|개인화|note|메모|personal|comment/,  "whyYou"],
  [/name|이름|creator|크리에이터/,           "creatorName"]
];
function normHeader(s){ return String(s||"").toLowerCase().replace(/[^a-z0-9가-힣]/g,""); }
function mapHeader(h){
  const n = normHeader(h);
  if(!n) return null;
  for(const [re,key] of COLRULES) if(re.test(n)) return key;
  return null;
}
function splitLine(line){
  if(line.indexOf("\t") >= 0) return line.split("\t");
  const out=[]; let cur="", q=false;
  for(let i=0;i<line.length;i++){
    const c=line[i];
    if(c === '"'){ if(q && line[i+1]==='"'){ cur+='"'; i++; } else q=!q; }
    else if(c === "," && !q){ out.push(cur); cur=""; }
    else cur+=c;
  }
  out.push(cur); return out;
}
function parseBulk(){
  const raw = $("bulkText").value.trim();
  if(!raw) return { rows:[], bad:[] };
  const lines = raw.split(/\r?\n/).filter(l => l.trim());
  if(!lines.length) return { rows:[], bad:[] };

  // 첫 줄이 헤더인지: 수신주소 열이 잡히고, 그 칸 자체가 이메일이 아니어야 한다
  let cols = null, start = 0;
  const firstCells = splitLine(lines[0]).map(s => s.trim().replace(/^"|"$/g,""));
  const mapped = firstCells.map(mapHeader);
  const looksHeader = mapped.some(k => k === "to") &&
                      !firstCells.some(c => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(c));
  if(looksHeader){ cols = mapped; start = 1; }

  const rows = [], bad = [];
  for(let i=start;i<lines.length;i++){
    const cells = splitLine(lines[i]).map(s => s.trim().replace(/^"|"$/g,""));
    const r = {};
    if(cols){
      cols.forEach((k,j) => { if(k && cells[j]) r[k] = cells[j]; });
    } else {
      // 헤더가 없으면 이메일처럼 생긴 칸을 찾고 나머지 첫 칸을 이름으로
      cells.forEach(c => {
        if(!r.to && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(c)) r.to = c;
        else if(!r.creatorName && c && !/^[0-9.%]+$/.test(c)) r.creatorName = c;
      });
    }
    if(!r.to){ if(lines[i].trim()) bad.push({ line:i+1, text:lines[i].slice(0,60), why:"이메일 없음" }); continue; }
    if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(r.to)){ bad.push({ line:i+1, text:r.to, why:"이메일 형식 오류" }); continue; }
    rows.push(r);
  }
  // 중복 제거 (같은 사람에게 두 번 나가는 사고 방지)
  const seen = new Set(), uniq = [];
  rows.forEach(r => {
    const k = r.to.toLowerCase();
    if(seen.has(k)){ bad.push({ line:"-", text:r.to, why:"중복 — 건너뜀" }); return; }
    seen.add(k); uniq.push(r);
  });
  return { rows:uniq, bad };
}
function renderBulkTable(){
  const { rows, bad } = parseBulk();
  $("bulkInfo").innerHTML = rows.length || bad.length
    ? '<div><b>' + rows.length + '</b> 명 인식' + (bad.length ? ' · <span style="color:#c0392b"><b>'+bad.length+'</b> 행 제외</span>' : '') + '</div>'
    : '';
  if(!rows.length && !bad.length){ $("bulkTable").style.display="none"; return; }

  let h = '<table class="rcp"><thead><tr><th>#</th><th>이메일</th><th>이름</th><th>핸들</th><th>금액</th><th>커미션</th><th>개인화</th></tr></thead><tbody>';
  rows.forEach((r,i) => {
    const p = heldPriorOf(r.to);   // 다른 담당자가 보낸 경우만 보류 표시 (본인 재발송은 그대로 나감)
    h += '<tr'+(p ? ' style="background:#fffaf0"' : '')+'><td>'+(i+1)+'</td><td>'+esc(r.to)+
         (p ? ' <span class="holdtag" title="'+esc(priorText(p))+'">⏸ 보류</span>' : '') +
         '</td><td>'+esc(r.creatorName||'')+'</td><td>'+esc(r.handle||'')+
         '</td><td>'+esc(r.amount||'')+'</td><td>'+esc(r.commission||'')+'</td><td>'+esc((r.whyYou||'').slice(0,40))+'</td></tr>';
  });
  bad.forEach(b => {
    h += '<tr><td class="bad">–</td><td class="bad" colspan="6">'+esc(b.text)+' — '+esc(b.why)+' (줄 '+b.line+')</td></tr>';
  });
  h += '</tbody></table>';
  $("bulkTable").innerHTML = h;
  $("bulkTable").style.display = "";
}

/* ── 제품 이미지 파일 첨부 ──
   상품이 TikTok Shop 목록에 없어 이미지 URL 이 없을 때 파일을 그대로 담아 보낸다.
   보내기 전에 브라우저에서 최대 1000px 로 줄이고 JPEG 로 다시 인코딩한다 —
   원본을 그대로 실으면 요청 본문이 수 MB 가 돼 Vercel 한도를 넘고 발송이 느려진다. */
const IMG_MAX_PX = 1000;
function loadImageFile(file){
  return new Promise((resolve, reject) => {
    const fr = new FileReader();
    fr.onerror = () => reject(new Error("파일을 읽지 못했습니다"));
    fr.onload = () => {
      const img = new Image();
      img.onerror = () => reject(new Error("이미지를 열지 못했습니다"));
      img.onload = () => {
        const scale = Math.min(1, IMG_MAX_PX / Math.max(img.width, img.height));
        const w = Math.max(1, Math.round(img.width * scale)), h = Math.max(1, Math.round(img.height * scale));
        const cv = document.createElement("canvas"); cv.width = w; cv.height = h;
        const cx = cv.getContext("2d");
        cx.fillStyle = "#ffffff"; cx.fillRect(0, 0, w, h);   // 투명 배경은 흰색으로 (JPEG 는 알파가 없다)
        cx.drawImage(img, 0, 0, w, h);
        resolve(cv.toDataURL("image/jpeg", 0.85));
      };
      img.src = fr.result;
    };
    fr.readAsDataURL(file);
  });
}

/* ── 중복 발송 확인 ──
   담당자가 10명이면 명단이 겹치는 건 시간 문제라, 명단을 넣는 즉시 이미 보낸 사람을
   표시한다. 실제 차단은 서버가 발송 직전에 다시 한다 — 확인과 발송 사이에 다른
   담당자가 먼저 보낼 수 있고, 그 경합은 서버에서만 막을 수 있기 때문. */
let DUP = {};            // 정규화한 주소 → 이전 발송 기록
let dupTimer = null;
const dupKey = e => {
  const s = String(e||"").trim().toLowerCase();
  const at = s.lastIndexOf("@");
  if(at < 1) return s;
  let loc = s.slice(0,at); const dom = s.slice(at+1);
  const plus = loc.indexOf("+");            // a+tag@x 는 같은 편지함
  if(plus > 0) loc = loc.slice(0,plus);
  return loc + "@" + dom;
};
function priorOf(to){ return DUP[dupKey(to)] || null; }
/* 내 발신 주소 (accounts 모드면 로그인 계정, byo 면 저장한 계정) */
function myEmail(){ const a = acct(); return a && a.email ? String(a.email).toLowerCase() : ""; }
/* 보류 대상인가 — **다른 담당자**가 이미 보낸 경우만. 본인 재발송은 서버가 그대로 보내므로
   보류가 아니다 (미리보기가 "보류"라고 잘못 겁주지 않도록 여기서도 본인 것은 뺀다). */
function heldPriorOf(to){
  const p = priorOf(to);
  if(!p) return null;
  return String(p.by || "").toLowerCase() === myEmail() ? null : p;
}

function scheduleDupCheck(){
  if(!(CFG.history && CFG.history.enabled)) return;
  clearTimeout(dupTimer);
  dupTimer = setTimeout(checkDup, 600);
}

async function checkDup(){
  if(!(CFG.history && CFG.history.enabled)) return;
  const list = recipients();
  if(!list.length){ $("dupInfo").innerHTML = ""; return; }
  try {
    const r = await fetch("/api/history", {
      method:"POST", cache:"no-store",
      headers: Object.assign({ "content-type":"application/json" }, authHeaders()),
      body: JSON.stringify({ recipients: list.map(x => ({ to:x.to, handle:x.handle||"" })) })
    });
    if(r.status === 401) return;
    const d = await r.json();
    if(d.error) throw new Error(d.error);
    DUP = {};
    (d.results||[]).forEach(x => { if(x.prior) DUP[dupKey(x.to)] = x.prior; });
    paintDup();
    if(MODE === "bulk") renderBulkTable();
  } catch(e){
    $("dupInfo").innerHTML = '<div class="muted" style="margin-top:8px">이력 확인 실패 — '+esc(String(e.message||e))+'</div>';
  }
}

function priorText(p){
  if(!p) return "";
  const who = p.byName || p.by || "다른 담당자";
  const when = p.at ? String(p.at).slice(0,10) : "";
  return who + (when ? " · " + when : "") + (p.campaign ? " · " + p.campaign : "");
}

function paintDup(){
  const list = recipients();
  const held = list.filter(r => heldPriorOf(r.to));   // 다른 담당자가 보낸 경우만 보류
  if(!held.length){ $("dupInfo").innerHTML = ""; return; }
  const forced = $("forceSend").checked;
  $("dupInfo").innerHTML =
    '<div class="' + (forced ? "err" : "hold") + '" style="margin-top:10px">' +
      (forced
        ? '⚠️ 다른 담당자가 이미 보낸 <b>'+held.length+'명</b> 에게도 <b>강제 발송</b>합니다.'
        : '⏸ 다른 담당자가 이미 보낸 <b>'+held.length+'명</b> 은 보류됩니다 (나머지만 발송).') +
    '</div>' +
    '<div class="scroll"><table class="rcp"><thead><tr><th>수신자</th><th>이전 발송</th></tr></thead><tbody>' +
    held.slice(0,50).map(r => '<tr><td>'+esc(r.to)+'</td><td>'+esc(priorText(heldPriorOf(r.to)))+'</td></tr>').join("") +
    '</tbody></table></div>';
}

async function loadSchedules(){

/* 응답을 JSON 으로 읽되, JSON 이 아니면(플랫폼 오류 페이지 등) 상태코드와 본문 앞부분을
   담은 error 로 바꿔 준다. 504=시간 초과, 502/500=서버 오류, 413=본문이 너무 큼 등을 구분한다. */
async function parseSend(r){
  const text = await r.text();
  try { return JSON.parse(text); }
  catch(_){
    const snippet = text.replace(/<[^>]*>/g, " ").replace(/\s+/g, " ").trim().slice(0, 140);
    const hint = r.status === 504 ? "발송이 제한시간을 넘겼습니다 — 한 번에 보내는 인원을 줄이거나 잠시 후 다시 시도하세요"
      : r.status === 413 ? "요청이 너무 큽니다 — 제품 이미지를 더 작게 하거나 URL 방식으로 바꿔보세요"
      : (r.status >= 500 ? "서버 함수 오류 — 잠시 후 다시 시도하세요" : "");
    return { error: "서버 오류 (HTTP " + r.status + ")" + (snippet ? ": " + snippet : ""), hint };
  }
}

async function doSend(list){
  const a = acct();
  const chunkSize = CFG.maxPerRequest || 25;
  const camp = campaign();
  const all = [];
  let done = 0;

  $("sendBtn").disabled = true;
  $("sendResult").innerHTML = "";
  $("progWrap").style.display = "";
  $("progFill").style.width = "0%";

  for(let i=0; i<list.length; i+=chunkSize){
    const chunk = list.slice(i, i+chunkSize);
    try {
      const r = await fetch("/api/outreach-send", {
        method: "POST",
        headers: Object.assign({ "content-type":"application/json" }, authHeaders()),
        body: JSON.stringify(Object.assign(
          { campaign: camp, recipients: chunk, force: $("forceSend").checked },
          // accounts 모드에서는 서버가 로그인 세션으로 발신 계정을 정한다.
          // 앱 비밀번호가 브라우저에 없으므로 보낼 것도 없다.
          AUTH_MODE === "accounts" ? {}
            : { auth: { email:a.email, password:a.password, name:a.name, title:a.title } }
        ))
      });
      if(r.status === 401){ showGate(AUTH_MODE === "accounts" ? "다시 로그인해 주세요." : "비밀번호가 올바르지 않습니다."); break; }
      // 서버가 JSON 이 아닌 오류 페이지(크래시·타임아웃 등)를 줄 수도 있다. 그럴 때
      // "not valid JSON" 대신 실제 상태코드와 본문 앞부분을 보여줘 원인을 알 수 있게 한다.
      const data = await parseSend(r);
      if(data.error){
        chunk.forEach(c => all.push({ to:c.to, ok:false, error:data.error + (data.hint ? " — " + data.hint : "") }));
      } else {
        (data.results||[]).forEach(x => all.push(x));
      }
    } catch(e){
      chunk.forEach(c => all.push({ to:c.to, ok:false, error:String(e.message||e) }));
    }
    done += chunk.length;
    $("progFill").style.width = Math.round(done/list.length*100) + "%";
    paintResults(all, list.length);
  }

  paintResults(all, list.length, true);
  $("sendBtn").disabled = false;
  checkDup();   // 방금 보낸 사람들이 이력에 잡히도록 다시 조회
}

function paintResults(results, total, final){
  const ok = results.filter(r => r.ok).length;
  const held = results.filter(r => !r.ok && r.held);   // 보류 — 실패가 아니라 의도된 차단
  const bad = results.filter(r => !r.ok && !r.held);
  let h = "";
  if(final){
    if(bad.length){
      h += '<div class="err"><b>'+ok+'/'+total+'</b> 발송 완료 · <b>'+bad.length+'건 실패</b>' +
           (held.length ? ' · '+held.length+'건 보류' : '') + '</div>';
    } else if(held.length){
      h += '<div class="hold">✅ <b>'+ok+'명</b> 발송 완료 · ⏸ <b>'+held.length+'명</b> 은 이미 발송돼 보류했습니다.</div>';
    } else {
      h += '<div class="ok">✅ <b>'+ok+'명</b> 발송 완료</div>';
    }
  } else {
    h += '<div class="muted" style="margin:6px 0">'+results.length+'/'+total+' 처리 중…</div>';
  }
  if(bad.length){
    h += '<div class="scroll"><table class="rcp"><thead><tr><th>수신자</th><th>사유</th></tr></thead><tbody>' +
      bad.map(b => '<tr><td>'+esc(b.to)+'</td><td class="bad">'+esc(b.error||"")+'</td></tr>').join("") +
      '</tbody></table></div>';
  }
  if(held.length && final){
    h += '<div class="muted" style="margin-top:10px">' + (CFG.admin
        ? '보류된 수신자 — 그래도 보내려면 [강제 발송] 을 켜고 다시 누르세요.'
        : '이미 발송된 주소라 보류되었습니다. 중복 발송은 관리자만 할 수 있고, 이 시도는 관리자 페이지에 남습니다.') + '</div>' +
      '<div class="scroll"><table class="rcp"><thead><tr><th>수신자</th><th>이전 발송</th></tr></thead><tbody>' +
      held.map(b => '<tr><td>'+esc(b.to)+'</td><td>'+esc(priorText(b.prior) || b.error || "")+'</td></tr>').join("") +
      '</tbody></table></div>';
  }
  $("sendResult").innerHTML = h;
}

function fmtAt(s){
  if(!s) return "";
  const t = Date.parse(s);
  if(!isFinite(t)) return String(s).slice(0,10);
  return new Date(t).toISOString().slice(0,10);
}

/* ── 중복 검사 (여러 개 한 번에) ──
   핸들/이메일을 한 줄에 하나씩(또는 콤마) 붙여넣으면 전 담당자(관리자 포함) 발송 이력을
   한 번에 조회한다. 서버는 로그를 딱 한 번 읽어 전부 대조한다(/api/lookup POST). */
let CHECK_DATA = null;
let CHECK_FILTER = "all";   // all | dup | clean

function parseChecks(text){
  // 줄바꿈·콤마·세미콜론·공백으로 나눈다 (핸들·이메일엔 공백이 없다)
  const seen = new Set(), out = [];
  String(text||"").split(/[\s,;]+/).forEach(s => {
    const v = s.trim(); if(!v) return;
    const k = v.toLowerCase();
    if(seen.has(k)) return; seen.add(k); out.push(v);
  });
  return out;
}
function paintCheckCount(){
  const n = parseChecks($("checkQ") ? $("checkQ").value : "").length;
  if($("checkCount")) $("checkCount").textContent = n ? (n + "개 입력됨") : "";
}

async function runCheck(){
  const items = parseChecks($("checkQ").value);
  const box = $("checkResult");
  if(!items.length){ box.innerHTML = '<div class="muted">핸들 또는 이메일을 입력하세요.</div>'; return; }
  box.innerHTML = '<div class="muted">'+items.length+'개 조회 중…</div>';
  try {
    const r = await fetch("/api/lookup", {
      method:"POST", cache:"no-store",
      headers: Object.assign({ "content-type":"application/json" }, authHeaders()),
      body: JSON.stringify({ queries: items })
    });
    if(r.status === 401){ showGate("다시 로그인해 주세요."); return; }
    const d = await parseSend(r);
    if(d.error) throw new Error(d.error + (d.hint ? " — " + d.hint : ""));
    CHECK_DATA = d; CHECK_FILTER = "all";
    renderChecks();
  } catch(e){
    box.innerHTML = '<div class="err">'+esc(String(e.message||e))+'</div>';
  }
}

  try { await navigator.clipboard.writeText(list.join("\n"));
    const el = $("copyMsg"); if(el){ el.textContent = "복사됨 ("+list.length+"개)"; setTimeout(()=>{ if(el) el.textContent=""; }, 2000); }
  } catch(e){ alert("복사 실패: "+(e.message||e)); }
}

    $("imgUpBtn").addEventListener("click", () => $("imgUpFile").click());
    $("imgUpFile").addEventListener("change", onImgFile);
    $("imgUpClear").addEventListener("click", () => setImgData(""));
    // 자동 리마인드 토글 — 켜면 주기·횟수 입력칸이 보인다. 상태를 브라우저에 저장한다.
    try { if(localStorage.getItem("outreach_remind_on") === "1") $("remindOn").checked = true; } catch(e){}
    const paintRemind = () => {
      const on = $("remindOn").checked;
      $("remindOpts").style.display = on ? "inline-flex" : "none";
      $("remindMsgs").style.display = on ? "block" : "none";
      try { localStorage.setItem("outreach_remind_on", on ? "1" : "0"); } catch(e){}
    };
    $("remindOn").addEventListener("change", paintRemind);
    paintRemind();

    // 하단 추가 안내 박스 — 체크하면 입력칸이 보이고, 상태를 브라우저에 저장한다.
    ["extra1","extra2"].forEach(k => {
      try { if(localStorage.getItem("outreach_"+k+"_on") === "1") $(k+"On").checked = true; } catch(e){}
      const paint = () => {
        const on = $(k+"On").checked;
        $(k+"Box").style.display = on ? "block" : "none";
        try { localStorage.setItem("outreach_"+k+"_on", on ? "1" : "0"); } catch(e){}
        renderPreview();
      };
      $(k+"On").addEventListener("change", paint);
      paint();
    });
  }

  // 관리자 페이지는 관리자만 볼 수 있다 — 버튼도 관리자에게만 띄운다.
  // (로그인 뒤에도 갱신돼야 하므로 최초 1회 블록 밖에 둔다)
  $("adminNav").innerHTML = CFG.admin
    ? ' <a class="btn" href="/admin.html">📊 관리자 페이지</a>' : "";

  // 강제 발송(중복 무시)은 관리자만 할 수 있다. 그 외 담당자에게는 체크박스를 아예 안 띄우고,
  // 중복은 자동 보류돼 관리자 페이지에 남는다. (서버도 관리자가 아니면 force 를 무시한다)
  if(CFG.history && CFG.history.enabled && CFG.admin){
    $("forceWrap").style.display = "flex";
    $("forceWrap").title = CFG.history.windowDays + "일 안에 이미 보낸 주소는 자동 보류됩니다";
  } else {
    $("forceWrap").style.display = "none";
    if($("forceSend").checked) $("forceSend").checked = false;   // 혹시 남아 있던 체크 해제
  }

  // 자동 리마인드는 이력 저장소가 있어야 동작한다(예약을 저장할 곳이 필요). 관리자 여부와 무관.
  // block 으로 둔다 — 안에 [토글 줄] + [회차별 문구]가 세로로 쌓여야 한다(flex 면 옆으로 붙는다).
  $("remindWrap").style.display = (CFG.history && CFG.history.enabled) ? "block" : "none";

  // 예약 발송 — 로그인 방식(서버가 앱 비밀번호 보유) + 이력 저장소가 있어야 동작한다.
  const canSched = AUTH_MODE === "accounts" && CFG.history && CFG.history.enabled;
  $("schedWrap").style.display = canSched ? "block" : "none";
  if(canSched){
    if(!$("schedDate").value){
      const t = new Date(Date.now() + 86400e3);   // 기본값: 내일
      $("schedDate").value = t.toISOString().slice(0,10);
    }
    if(!$("schedWrap").dataset.wired){
      $("schedWrap").dataset.wired = "1";
      $("schedOn").addEventListener("change", paintSched);
      paintSched();
    }
    loadSchedules();
  }

  // 상단 화면 전환 탭.
  //  · Mail Pipeline — 로그인 방식(본인 식별) + 이력 저장소가 있어야 (본인 것만 보여주므로)
  //  · 중복 검사 — 이력 저장소만 있으면 (BYO 방식이어도 공용 이력을 조회할 수 있다)
  const canPipe = AUTH_MODE === "accounts" && CFG.history && CFG.history.enabled;
  const canCheck = Boolean(CFG.history && CFG.history.enabled);
  $("vtabPipeline").style.display = canPipe ? "" : "none";
  $("vtabCheck").style.display = canCheck ? "" : "none";
  $("viewTabs").style.display = (canPipe || canCheck) ? "flex" : "none";
  if(!canPipe && VIEW === "pipeline") setView("send");
  if(!canCheck && VIEW === "check") setView("send");
  PIPE_LOADED = false;

  // 상품 선택 버튼은 TikTok Shop 자격증명이 있을 때만 띄운다.
  // probe 는 설정 여부만 확인하므로 페이지를 열 때마다 상품을 조회하지 않는다.
  if(authed){
    fetch("/api/products?probe=1", { cache:"no-store", headers: authHeaders() })
      .then(r => r.json())
      .then(d => { if(d && d.ready) $("pickBar").style.display = "flex"; })
      .catch(() => {});

    // 영상 시트도 마찬가지 — 다만 안 되는 이유는 화면에 남긴다
    fetch("/api/videos?probe=1", { cache:"no-store", headers: authHeaders() })
      .then(r => r.json())
      .then(d => {
        VIDEOS_READY = Boolean(d && d.ready);
        VIDEOS_WHY = VIDEOS_READY ? "" : String((d && (d.reason || d.error)) || "시트를 읽을 수 없습니다");
        paintVidBar();
      })
      .catch(e => { VIDEOS_WHY = String(e.message||e); paintVidBar(); });
  }
  paintWho();
  renderPreview();
  if(authed && AUTH_MODE !== "accounts" && !acct()) openAcct();
}

boot();
